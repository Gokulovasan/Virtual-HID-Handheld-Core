cmake_minimum_required(VERSION 3.14)
project(Virtual-HID-Handheld-Core VERSION 1.0.0 LANGUAGES C)

# Set C standard
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

# Compiler flags for strict quality and industrial standards
if(MSVC)
    add_compile_options(/W4 /WX)
else()
    add_compile_options(-Wall -Wextra -Wpedantic -Werror -O2)
endif()

# Include directories
include_directories(include)

# Executable target
add_executable(virtual_hid_core
    src/main.c
)

# Installation target
install(TARGETS virtual_hid_core DESTINATION bin)
